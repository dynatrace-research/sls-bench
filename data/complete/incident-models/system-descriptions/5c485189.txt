sys:
  id: hosted_package_repo
  desc: >
    A hosted package repository serving metadata and package artifacts over HTTPS.
    Requests arrive through a load balancer and are handled by a small pool of frontend
    application servers. Frontends perform DNS lookups (via the datacenter resolver) to
    reach internal upstreams; during extreme scheduled automation spikes, DNS connectivity
    becomes a bottleneck and causes widespread request timeouts.

states: {n: normal, f: failure}

components:
  - id: external_monitor
    name: External Monitoring
    svc: monitor
    hosts: [mon-1]
    desc: >
      External uptime checker that probes the repository and pages on-call when checks fail.

    to:
      - dst: edge_lb
        proto: https
        desc: Probes repository endpoints through the public load balancer.

    logs:
      check_ok:
        desc: A successful probe result.
        lvl: INFO
        msg: "check ok url={url} status={status} latency_ms={latency_ms}"
        vars:
          url: {k: ch, v: ["https://repo.example.com/health", "https://repo.example.com/repodata/index.json"]}
          status: {k: ch, v: ["200", "204"]}
          latency_ms: {k: i, v: [20, 800]}
      check_fail:
        desc: A failed probe result (timeout or non-2xx response).
        lvl: ERROR
        msg: "check failed url={url} error={error} latency_ms={latency_ms}"
        vars:
          url: {k: ch, v: ["https://repo.example.com/health", "https://repo.example.com/repodata/index.json"]}
          error: {k: ch, v: ["timeout", "http_503", "http_504", "connection_reset"]}
          latency_ms: {k: i, v: [200, 8000]}
      page_sent:
        desc: Monitor pages the on-call team for the repository.
        lvl: CRITICAL
        msg: "paging oncall service={service} incident={incident_id}"
        vars:
          service: {k: ch, v: ["hosted-package-repo"]}
          incident_id: {k: hex, v: 12}

    beh:
      n:
        desc: Monitoring runs continuously but only emits logs via probe flows.
        emit: []
      f:
        desc: Monitoring runs continuously but only emits logs via probe flows; paging is modeled as a one-shot.
        emit: []

  - id: edge_lb

      access_5xx:
        desc: Access log entry for upstream timeouts/unavailable responses.
        lvl: INFO
        msg: "access method={method} uri={uri} status={status} bytes={bytes} rt_ms={rt_ms} backend={backend_host} upstream_ms={upstream_ms}"
        vars:
          method: {k: ch, v: ["GET", "HEAD"]}
          uri: {k: str, v: "repo_request_path"}
          status: {k: ch, v: ["503", "504"]}
          bytes: {k: i, v: [0, 3000]}
          rt_ms: {k: i, v: [200, 20000]}
          backend_host: {k: ch, v: ["fe-1", "fe-2", "fe-3"]}

      lb_stats:

        lvl: INFO

        vars: {}
        state_vars:
          n:

    beh:

        emit:

            per_min: 1
            scope: per_host
      f:
        desc: Emits periodic stats during the spike; queues and 5xx counts rise.
        emit:
          - id: lb_stats
            per_min: 1
            scope: per_host

    to:
      - dst: dns_resolver

    logs:

assumptions:
  - >
    This model focuses on a single outage during an extreme scheduled traffic spike after additional frontend capacity existed,
    where the limiting factor shifted to DNS connectivity (frontends blocked on connect() to the datacenter resolver).
  - >
    To align with the observed connect() stalls and resolver connection-error counters, the frontend-to-resolver dependency is modeled
    as TCP. (Real DNS may use UDP in many cases; this incident model intentionally emphasizes the connect()-style failure mode.)
  - >
    Load balancer "max connections" tuning and the later DNS-cache rollout are out of scope for this specific scenario; we model
    the incident that ends while still failing (before the DNS cache mitigation).
  - >
    app_frontend.app_metrics.dns_timeouts_1m is modeled as a per-host rolling 1-minute count of resolver timeouts/stalls and is kept
    consistent with the (rare) WARN emission rate in normal and the frequent WARNs in failure.
  - >
    Frontends do not emit per-request access logs to keep log volume realistic; per-request visibility comes primarily from the
    load balancer access logs, while DNS symptoms are visible via periodic metrics and elevated WARNs.
  - >
    Paths are modeled as coarse logical traversals and omit explicit return hops to satisfy graph connectivity while keeping the
    main request family (monitor->LB->frontend; client->LB->frontend) clear.
  - >
    edge_lb.access_5xx rt_ms/upstream_ms domains are extended to 20000ms to cover failure-phase p95 latency hints after event
    latency multipliers (up to ~18000ms implied p95), avoiding inconsistent generation/clamping.
  - >
    Estimated normal-phase logs/min: ~827 (800 LB access + ~18 app metrics + ~0.2 app DNS warns + ~2 LB stats + ~1 DNS stats + ~6 monitor checks).
  - >
    Estimated total logs over a 24-minute scenario: ~69k (dominant contributor is LB access logs during spike), within the 20k–100k target.
  - >
    Key diagnostic anchors expected to appear: edge_lb.access_5xx (customer impact), edge_lb.lb_stats (queue/5xx shift),
    app_frontend.app_metrics and app_frontend.dns_timeout (DNS stall evidence), dns_resolver.dns_stats (corroborating resolver stress),
    and oncall_ops.strace_sample (confirmatory sampling).
