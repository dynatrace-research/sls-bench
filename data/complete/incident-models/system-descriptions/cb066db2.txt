sys:
  id: duo1_authentication_platform
  desc: >
    A deployment of the Duo authentication service that handles inbound authentication checks from customer
    applications. Requests enter through an edge proxy and are served by an authentication API tier that
    relies on a shared relational database. Under sustained load, application threads can queue while waiting
    for a database connection, increasing end-to-end latency and causing intermittent timeouts.

states: {n: normal, f: failure}

components:
  - id: edge_proxy
    name: Edge Proxy
    svc: duo-edge
    hosts: [edge-1, edge-2]
    desc: >
      Public entrypoint for customer authentication checks. Forwards requests to the authentication API and
      returns responses to clients. Operators can enable request limiting at this layer.
    to:
      - dst: auth_api
        proto: https
        desc: Routes /auth/v2/check requests to the authentication API tier.
    logs:
      access_ok:
        desc: Successful client response for an authentication check.
        lvl: INFO
        msg: "{method} {route} status={status} dur_ms={dur_ms} req_id={req_id} upstream=auth_api"
        vars:
          method: {k: ch, v: [POST]}
          route: {k: ch, v: [/auth/v2/check]}
          status: {k: i, v: [200, 200]}
          dur_ms: {k: i, v: [20, 3000]}
          req_id: {k: hex, v: 16}
      access_504:
        desc: Client-visible timeout when the upstream does not complete in time.
        lvl: WARN
        msg: "{method} {route} status=504 dur_ms={dur_ms} req_id={req_id} upstream=auth_api err=upstream_timeout"
        vars:
          method: {k: ch, v: [POST]}
          route: {k: ch, v: [/auth/v2/check]}
          dur_ms: {k: i, v: [2500, 9000]}
          req_id: {k: hex, v: 16}
      access_429:
        desc: Client-visible rejection when upstream begins proactively rejecting due to internal queue cap.
        lvl: WARN
        msg: "{method} {route} status=429 dur_ms={dur_ms} req_id={req_id} upstream=auth_api err=too_many_requests"
        vars:
          method: {k: ch, v: [POST]}

          dur_ms: {k: i, v: [5, 80]}
          req_id: {k: hex, v: 16}
      rate_limit_enabled:
        desc: Operator action marker enabling request limiting at the edge.
        lvl: INFO
        msg: "enabled request limiting: max_rpm={max_rpm} scope=duo1"

          - id: proxy_worker_stats

  - id: auth_api
    name: Core Authentication API
    svc: duo-auth
    hosts: [auth-1, auth-2, auth-3, auth-4]
    desc: >
      Handles authentication checks and consults the database for user/device/factor data. Under load, threads
      can queue while waiting for a database connection from a pool, increasing request duration and causing
      timeouts. Mitigations include flushing the wait queue and enforcing a maximum queue size.
    to:
      - dst: auth_db
        proto: jdbc
        desc: Reads and writes authentication-related records.
      - dst: edge_proxy
        proto: https
        desc: Returns HTTP responses to the edge proxy.
    logs:
      db_acquire_wait_warn:
        desc: Per-request warning indicating long wait for a DB connection and queued pressure.
        lvl: WARN
        msg: "db connection wait high: waited_ms={waited_ms} queue_depth={queue_depth} req_id={req_id}"
        vars:
          waited_ms: {k: i, v: [200, 6000]}
          queue_depth: {k: i, v: [50, 9000]}
          req_id: {k: hex, v: 16}
      auth_ok:
        desc: Authentication check completed successfully (allow/deny decision returned).
        lvl: INFO
        msg: "auth completed: result={result} dur_ms={dur_ms} req_id={req_id} user={user_id}"

          result: {k: ch, v: [allow, deny]}
          dur_ms: {k: i, v: [30, 7000]}
          req_id: {k: hex, v: 16}
          user_id: {k: uuid, v: null}
      auth_timeout:
        desc: Authentication check failed due to timing out while acquiring a DB connection.
        lvl: ERROR
        msg: "auth failed: error=db_acquire_timeout waited_ms={waited_ms} dur_ms={dur_ms} req_id={req_id}"

          req_id: {k: hex, v: 16}
      queue_depth_metric:

        lvl: INFO

        vars: {}
        state_vars:
          n:
            queue_depth: {k: i, v: [0, 40]}
            inflight: {k: i, v: [10, 120]}
            p95_wait_ms: {k: i, v: [1, 30]}
          f:
            queue_depth: {k: i, v: [300, 9000]}
            inflight: {k: i, v: [80, 600]}
            p95_wait_ms: {k: i, v: [200, 7000]}
      db_pool_metric:
        desc: Periodic snapshot of DB connection pool usage and waiter count.
        lvl: INFO
        msg: "metric db_pool: in_use={in_use} max={max} waiters={waiters}"
        vars:
          max: {k: i, v: [200, 200]}
        state_vars:
          n:
            in_use: {k: i, v: [20, 120]}
            waiters: {k: i, v: [0, 10]}

            in_use: {k: i, v: [150, 200]}
            waiters: {k: i, v: [50, 1500]}
      queue_flushed:
        desc: Operator action marker flushing accumulated DB-wait queue on an application server.
        lvl: INFO
        msg: "flushed pending db-wait queue: dropped={dropped} reason={reason}"
        vars:
          dropped: {k: i, v: [500, 6000]}
          reason: {k: ch, v: [manual_mitigation]}
      queue_limit_enabled:
        desc: Operator action marker enabling a max queue limit to reject overflow instead of queuing.
        lvl: INFO
        msg: "set db-wait queue max={max_queue} action=reject_overflow"
        vars:
          max_queue: {k: i, v: [2000, 2000]}
    beh:
      n:
        desc: Regular metrics show low queue depth and minimal DB connection waiting.
        emit:
          - id: queue_depth_metric
            per_min: 1.0
          - id: db_pool_metric
            per_min: 1.0
      f:
        desc: Regular metrics show elevated queue depth and high DB pool usage under capacity stress.
        emit:
          - id: queue_depth_metric
            per_min: 1.0
          - id: db_pool_metric
            per_min: 1.0

  - id: auth_db
    name: Authentication Database
    svc: duo-mysql

    logs:

        state_vars:

            threads: {k: i, v: [10, 80]}
            qps: {k: i, v: [300, 1200]}

            active: {k: i, v: [600, 980]}
            threads: {k: i, v: [80, 300]}
            qps: {k: i, v: [600, 2200]}

    beh:
      n:

        emit:

assumptions:
  - |
    Names are genericized to three core components (edge proxy, auth API, DB) to preserve the incident’s causal core:
    DB-connection wait queuing and its mitigation.
  - |
    "Request limiting was ineffective" is modeled as reduced inbound request rate at the edge while the auth API continues
    to show heavy DB-wait queuing and timeouts until queues are flushed.
  - |
    Push/SMS/phone delivery impacts are not separately modeled; the incident is represented via the shared authentication
    path whose degradation explains customer-visible latency/timeouts.
  - |
    Normal-phase log volume estimate: ~550 rpm * 2 logs/request + ~12 background logs/min ≈ ~1100 logs/min.
  - |
    Total scenario log volume estimate (with failure multipliers in the scenario): ~77k logs (in the required 20k–100k range),
    dominated by per-request access/result logs plus per-minute metrics.
